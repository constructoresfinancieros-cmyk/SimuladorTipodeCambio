async function obtenerTasas() { const status = document.getElementById('status-text'); // Lista de APIs de respaldo (Fuentes distintas) const fuentes = [ 'https://ve.dolarapi.com/v1/dolares', 'https://pydolarve.org/api/v1/dollar?page=bcv' // Respaldo 1 ]; try { status.innerText = "Sincronizando datos..."; const response = await fetch(fuentes[0]); if (!response.ok) throw new Error("API Principal caída"); const data = await response.json(); // Buscamos los nombres (manejando variaciones de la API) const dolarBCV = data.find(d => d.nombre === 'Oficial' || d.nombre === 'BCV'); const dolarPara = data.find(d => d.nombre === 'Paralelo' || d.nombre === 'EnParaleloVzla'); bcv = dolarBCV.promedio; paralelo = dolarPara.promedio; actualizarInterfaz(); status.innerText = "Sincronización Automática Activa"; } catch (error) { console.error("Fallo fuente principal, intentando respaldo..."); // Aquí podrías añadir un segundo fetch a otra API si tuvieras el endpoint listo status.innerText = "Error de servidor externo. Reintentando en breve..."; setTimeout(obtenerTasas, 30000); // Reintenta cada 30 seg si está caído } } function actualizarInterfaz() { document.getElementById('tasa-bcv').innerText = bcv.toFixed(2) + " Bs"; document.getElementById('tasa-binance').innerText = paralelo.toFixed(2) + " Bs"; calcular(); }
